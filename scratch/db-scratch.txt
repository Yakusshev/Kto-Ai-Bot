-- ШАГ 1: Создаем временную таблицу с ключами для удаления
$keys_to_delete = (
    SELECT
        -- Создаем уникальный строковый ключ
        CAST(chat_id AS String) || '|' ||
        CAST(message_id AS String) || '|' ||
        CAST(created_at AS String) AS composite_key
    FROM (
        SELECT
            chat_id,
            message_id,
            created_at,
            ROW_NUMBER() OVER (
                PARTITION BY chat_id, message_id
                ORDER BY created_at ASC
            ) AS rn
        FROM chat_events_release
        WHERE chat_id = 7659801120
    )
    WHERE rn > 1
);

-- ШАГ 2: Удаляем по ключу
DELETE FROM chat_events_release
WHERE
    CAST(chat_id AS String) || '|' ||
    CAST(message_id AS String) || '|' ||
    CAST(created_at AS String) IN (
        SELECT composite_key FROM $keys_to_delete
    );
