CREATE TABLE chat_events (
    chat_id Uint64,
    created_at Timestamp,
    message_id Uint64,
    event_type Utf8,    -- 'text', 'voice', 'photo', 'sticker', 'reply'
    role Utf8,          -- 'user', 'assistant'
    content Utf8,       -- Текст или описание медиа
    reply_to_id Uint64, -- ID сообщения, на которое ответили
    metadata Json,      -- Сюда можно кидать всё остальное (ссылку на фото, эмодзи стикера)
    PRIMARY KEY (chat_id, created_at)
);
